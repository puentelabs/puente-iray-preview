cmake_minimum_required(VERSION 3.22)
project(PuenteIrayPreview LANGUAGES CXX)
option(PUENTE_ENABLE_VALIDATION_API "Include local event-loop test helper" OFF)
option(PUENTE_PACKAGED_ONLY "Require bundled worker; never use external Python" ON)
set(DAZ_SDK_DIR "" CACHE PATH "Official Daz Studio 6 SDK root")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This prototype targets Windows x64 Daz Studio 6")
endif()
foreach(required include/dzpane.h x64/lib/dzcore.lib x64/lib/dzsdkmemory.lib)
  if(NOT EXISTS "${DAZ_SDK_DIR}/${required}")
    message(FATAL_ERROR "Missing Studio 6 SDK file: ${DAZ_SDK_DIR}/${required}. Set DAZ_SDK_DIR.")
  endif()
endforeach()
find_package(Qt6 6.10.3 EXACT REQUIRED COMPONENTS Core Core5Compat Gui Widgets Qml)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_library(PuenteIrayPreview SHARED PreviewPane.cpp RenderTools.cpp RenderTools.h WorkerRuntime.cpp WorkerRuntime.h PuenteIrayPreview.def
  "${DAZ_SDK_DIR}/include/dzsdkmemorydefines.cpp")
target_include_directories(PuenteIrayPreview PRIVATE "${DAZ_SDK_DIR}/include")
target_compile_definitions(PuenteIrayPreview PRIVATE DZ_QT_MIGRATION)
if(PUENTE_PACKAGED_ONLY)
  target_compile_definitions(PuenteIrayPreview PRIVATE PUENTE_PACKAGED_ONLY)
endif()
if(PUENTE_ENABLE_VALIDATION_API)
  target_compile_definitions(PuenteIrayPreview PRIVATE PUENTE_ENABLE_VALIDATION_API)
endif()
target_compile_options(PuenteIrayPreview PRIVATE /MD /U_DEBUG /Zc:__cplusplus /permissive-)
target_link_libraries(PuenteIrayPreview PRIVATE Qt6::Core Qt6::Core5Compat
  Qt6::Gui Qt6::Widgets Qt6::Qml "${DAZ_SDK_DIR}/x64/lib/dzcore.lib"
  "${DAZ_SDK_DIR}/x64/lib/dzsdkmemory.lib")
set_target_properties(PuenteIrayPreview PROPERTIES PREFIX "" OUTPUT_NAME "dsp_PuenteIrayPreview")
# No automatic installation into Daz. First build must be checked against the
# matching SDK/app pair; Studio 6 SDK versions can change binary compatibility.
